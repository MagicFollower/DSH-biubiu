{{- /* Return the pre-order reading sequence represented by the sidebar tree. The root section index precedes its descendants; data/docs_nav.json is authoritative when the docs shell already uses that explicit tree. */ -}} {{- $page := .page -}} {{- $root := .root -}} {{- $pages := slice -}} {{- $docsNavSections := slice -}} {{- if and (in (slice "docs" "book") $page.Type) (ne $root $page.Site.Home) -}} {{- $docsNavSections = partialCached "shell/docs-nav-sections.html" (dict "page" $page "root" $root) $root.RelPermalink $page.Site.Language.Lang -}} {{- end -}} {{- if $docsNavSections -}} {{- /* A real docs section index is part of the reading sequence. A site home used only as a book/landing cover is not silently prepended. */ -}} {{- if partial "shell/nav-page-enabled.html" $root -}} {{- $pages = $pages | append $root -}} {{- end -}} {{- range $docsNavSections -}} {{- $branch := partial "shell/nav-flatten-node.html" (dict "page" $page "node" .) -}} {{- range $branch }}{{ $pages = $pages | append . }}{{ end -}} {{- end -}} {{- else -}} {{- $includeRoot := or (ne $root $page.Site.Home) (eq $root.Type $page.Type) -}} {{- $pages = partial "shell/nav-flatten-section.html" (dict "section" $root "includeSelf" $includeRoot ) -}} {{- end -}} {{- return ($pages | uniq) -}}
