{{- /* Headings, footnotes, and generated tab IDs are page-local. Namespace them and their same-page links only while assembling a multi-page Print document; explicit fig/tbl/eq and authored anchors remain byte-stable unless an author declares a page-local ID with data-td-print-local. */ -}} {{- $page := .page -}} {{- $content := printf "%s" .content -}} {{- $anchors := slice -}} {{- range $page.Fragments.Identifiers }}{{ $anchors = $anchors | append . }}{{ end -}} {{- range .headings }}{{ $anchors = $anchors | append . }}{{ end -}} {{- /* Grouped tabs deliberately share group/value IDs across HTML pages. Collect their rendered containers and panels before rewriting so nested and repeated peer groups retain the IDs chosen by the shortcode. This leaves ordinary HTML and single-page Print deep links unchanged. */ -}} {{- range $tag := findRE `]*class="(?:[^"]*\s)?td-tabs(?:__panel)?(?:\s[^"]*)?"[^>]*>` $content -}} {{- range findRESubmatch `\sid="([^"]+)"` $tag -}} {{- $anchors = $anchors | append (htmlUnescape (index . 1)) -}} {{- end -}} {{- end -}} {{- $tags := findRE `(?i)<[a-z][a-z0-9:-]*\s(?:[^>"']|"[^"]*"|'[^']*')*>` $content -}} {{- range $tag := $tags -}} {{- if strings.Contains (lower $tag) "data-td-print-local" -}} {{- $local := false -}} {{- $id := "" -}} {{- range $token := findRE `(?i)\s+[^\s"'<>/=]+(?:\s*=\s*(?:"[^"]*"|'[^']*'|[^\s>]+))?` $tag -}} {{- if findRE `(?i)^\s+data-td-print-local(?:\s*=|$)` $token }}{{ $local = true }}{{ end -}} {{- range $match := findRESubmatch `(?i)^\s+id\s*=\s*(?:"([^"]*)"|'([^']*)'|([^\s>]+))$` $token -}} {{- $id = htmlUnescape (or (index $match 1) (index $match 2) (index $match 3)) -}} {{- end -}} {{- end -}} {{- if and $local $id }}{{ $anchors = $anchors | append $id }}{{ end -}} {{- end -}} {{- end -}} {{- $hrefMap := newScratch -}} {{- range $anchor := uniq $anchors -}} {{- $target := partial "book/aggregate-heading-anchor.html" (dict "page" $page "anchor" $anchor) -}} {{- $hrefMap.SetInMap "ids" $anchor $target -}} {{- $targetHref := replace $target "%" "%25" -}} {{- $hrefMap.SetInMap "hrefs" (printf "#%s" $anchor) (printf "#%s" $targetHref) -}} {{- $hrefMap.SetInMap "hrefs" (printf "%s#%s" $page.RelPermalink $anchor) (printf "#%s" $targetHref) -}} {{- end -}} {{- /* Read actual attribute values: Go's HTML attribute escaping also encodes characters such as +, while Fragments contains the raw identifier. Token boundaries protect id-shaped text inside another attribute. */ -}} {{- with $ids := $hrefMap.Get "ids" -}} {{- range $tag := $tags -}} {{- if findRE `(?i)\sid\s*=` $tag -}} {{- $chunks := slice -}} {{- range $token := findRE `(?i)<[a-z][a-z0-9:-]*|\s+[^\s"'<>/=]+(?:\s*=\s*(?:"[^"]*"|'[^']*'|[^\s>]+))?|\s*/?>` $tag -}} {{- $chunk := $token -}} {{- range $match := findRESubmatch `(?i)^\s+id\s*=\s*(?:"([^"]*)"|'([^']*)'|([^\s>]+))$` $token -}} {{- $value := htmlUnescape (or (index $match 1) (index $match 2) (index $match 3)) -}} {{- with index $ids $value }}{{ $chunk = printf ` id="%s"` (htmlEscape .) }}{{ end -}} {{- end -}} {{- $chunks = $chunks | append $chunk -}} {{- end -}} {{- $resolved := delimit $chunks "" -}} {{- if ne $tag $resolved }}{{ $content = replace $content $tag $resolved }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $hrefs := $hrefMap.Get "hrefs" -}} {{- range $tag := findRE `(?i)"']|"[^"]*"|'[^']*')*>` $content -}} {{- /* Like print/resolve-links.html, consume whole attribute values. Rebuild from tokens so even identical href text inside title stays untouched. */ -}} {{- $chunks := slice -}} {{- range $token := findRE `(?i)/=]+(?:\s*=\s*(?:"[^"]*"|'[^']*'|[^\s>]+))?|\s*/?>` $tag -}} {{- $chunk := $token -}} {{- range $match := findRESubmatch `(?i)^\s+href\s*=\s*(?:"([^"]*)"|'([^']*)'|([^\s>]+))$` $token -}} {{- $value := htmlUnescape (or (index $match 1) (index $match 2) (index $match 3)) -}} {{- /* URL fragments use percent escaping, not form/query escaping: Chinese headings and encoded ASCII must match the raw ID, while a literal plus remains a plus. Keep the URL prefix byte-stable so external, queried, and other-page links retain their destination. Malformed escapes remain untouched instead of failing a build. */ -}} {{- if and (strings.Contains $value "%") (strings.Contains $value "#") -}} {{- $parsed := try (urls.Parse $value) -}} {{- if not $parsed.Err -}} {{- $value = printf "%s#%s" (index (split $value "#") 0) $parsed.Value.Fragment -}} {{- end -}} {{- end -}} {{- with index $hrefs $value -}} {{- $chunk = printf ` href="%s"` (htmlEscape .) -}} {{- end -}} {{- end -}} {{- $chunks = $chunks | append $chunk -}} {{- end -}} {{- $resolved := delimit $chunks "" -}} {{- if ne $tag $resolved -}} {{- $content = replace $content $tag $resolved -}} {{- end -}} {{- end -}} {{- end -}} {{/* Footnotes are numbered per page (`fn:1`, `fnref:1`, `fnref2:1` for a repeat reference), so every chapter that carries references would otherwise emit the same IDs into one print document. The page prefix goes after the kind so the IDs stay recognisable as footnotes. */ -}} {{- $prefix := partial "book/aggregate-heading-anchor.html" (dict "page" $page "anchor" "") -}} {{- $content = replaceRE `(id|href)="(#?)(fnref[0-9]*|fn):` (printf `${1}="${2}${3}:%s` $prefix) $content -}} {{- return ($content | safeHTML) -}}
