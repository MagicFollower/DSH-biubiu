{{- /* WCAG contrast ratio of a parsed color against a theme canvas. Input: `dict "color" "canvas" "light"|"dark"`. Returns the ratio as a float. The canvas luminances are precomputed from the site default preset's canvases -- Paper #f7f6f3 light, #161513 dark (in `_presets.scss`); Slate #f1f4f8 light, #0b1119 dark (in `_brand.scss`); Ink and Terminal from `_experimental-presets.scss`. check-presets.py verifies each luminance against the palette, so this is advisory against the theme's own defaults, not against a reader's other preset choice or whatever a consumer site may repaint. */ -}} {{- $color := .color -}} {{- $lum := 0.0 -}} {{- range slice (dict "v" $color.r "w" 0.2126) (dict "v" $color.g "w" 0.7152) (dict "v" $color.b "w" 0.0722) -}} {{- $cs := div (float .v) 255 -}} {{- $lin := cond (le $cs 0.04045) (div $cs 12.92) (math.Pow (div (add $cs 0.055) 1.055) 2.4) -}} {{- $lum = add $lum (mul .w $lin) -}} {{- end -}} {{- $preset := (partialCached "preset-config.html" "preset-global").preset -}} {{- $canvases := dict "paper" (dict "light" 0.921568 "dark" 0.007539) "slate" (dict "light" 0.901795 "dark" 0.005422) "ink" (dict "light" 1.000000 "dark" 0.003347) "terminal" (dict "light" 0.909487 "dark" 0.004515) -}} {{- $canvas := index $canvases $preset (cond (eq .canvas "dark") "dark" "light") -}} {{- $hi := math.Max $lum $canvas -}} {{- $lo := math.Min $lum $canvas -}} {{- return div (add $hi 0.05) (add $lo 0.05) -}}
