{{- /* One resource authority for HTML, Print, Markdown, and album covers. */ -}} {{- $page := . -}} {{- $config := partialCached "expo/config.html" . .RelPermalink -}} {{- $items := slice -}} {{- $ids := slice -}} {{- $matched := try (.Resources.Match $config.images) -}} {{- if $matched.Err }}{{ warnf "expo.images: invalid glob at %s; using images/*" .Path }}{{ $matched = try (.Resources.Match "images/*") }}{{ end -}} {{- range $matched.Value -}} {{- if not (reflect.IsImageResource .) }}{{ continue }}{{ end -}} {{- $resource := . -}} {{- $media := partial "_funcs/media-result.html" (dict "resource" .) -}} {{- $original := $media -}} {{- $meta := .Params -}} {{- if isset $meta "original" -}} {{- $origVal := index $meta "original" -}} {{- $candidate := false -}} {{- if and (eq (printf "%T" $origVal) "string") (ne (strings.TrimSpace $origVal) "") }}{{ $candidate = $page.Resources.Get $origVal }}{{ end -}} {{- if and $candidate (reflect.IsImageResource $candidate) -}} {{- $original = partial "_funcs/media-result.html" (dict "resource" $candidate) -}} {{- else }}{{ warnf "expo: original %v must name a bundled image at %s; using the displayed resource" $origVal $page.Path }}{{ end -}} {{- end -}} {{- $title := .Title -}} {{- if eq $title .Name }}{{ $title = humanize (path.BaseName .Name) }}{{ end -}} {{- $values := dict "alt" ($media.alt | default $title) "caption" "" "byline" $media.byline -}} {{- range slice "alt" "caption" "byline" -}} {{- $key := . -}} {{- if isset $meta $key -}} {{- $value := index $meta $key -}} {{- if eq (printf "%T" $value) "string" }}{{ $values = merge $values (dict $key $value) }}{{ else }}{{ warnf "expo: resource %s %s must be a string at %s; ignoring it" $resource.Name $key $page.Path }}{{ end -}} {{- end -}} {{- end -}} {{- $hashID := printf "img-%s" (substr (crypto.SHA256 .Name) 0 16) -}} {{- $id := $hashID -}} {{- if isset $meta "id" -}} {{- $rawID := index $meta "id" -}} {{- $idType := printf "%T" $rawID -}} {{- $idStr := printf "%v" $rawID -}} {{- if and (or (eq $idType "string") (findRE `^u?int` $idType)) (findRE `^[A-Za-z0-9][A-Za-z0-9_-]*$` $idStr) -}} {{- $id = printf "img-%s" $idStr -}} {{- else }}{{ warnf "expo: id must contain ASCII letters, digits, underscore or hyphen at %s; using a filename hash" $page.Path }}{{ end -}} {{- end -}} {{- if in $ids $id -}} {{- warnf "expo: duplicate image id %q at %s; using a filename hash" $id $page.Path -}} {{- $id = $hashID -}} {{- if in $ids $id -}} {{- /* At most len(ids) suffixes can already be occupied. Check each candidate, including collisions with authored hash-like IDs. */ -}} {{- range seq 1 (add (len $ids) 1) -}} {{- $id = printf "%s-%d" $hashID . -}} {{- if not (in $ids $id) }}{{ break }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- $ids = $ids | append $id -}} {{- $weight := partial "validate.html" (dict "value" $meta.weight "kind" "number" "fallback" 0 "key" "expo resource weight" "where" $page.Path) -}} {{- $poster := $media.src -}} {{- if and $media.processable (or (gt $media.width 960) (gt $media.height 960) (eq $media.mediaType "image/gif")) -}} {{- $thumb := try (.Fit "960x960 webp") -}} {{- if $thumb.Err }}{{ warnf "expo: thumbnail failed for %s at %s: %s; using original" .Name $page.Path $thumb.Err }}{{ else }}{{ $poster = $thumb.Value.RelPermalink }}{{ end -}} {{- end -}} {{- $animated := eq $original.mediaType "image/gif" -}} {{- $src := $poster -}} {{- if $animated }}{{ $src = $original.src }}{{ end -}} {{- $items = $items | append (dict "id" $id "title" $title "alt" ($values.alt | default $title) "caption" $values.caption "byline" $values.byline "weight" $weight "src" $src "poster" $poster "original" $original.src "width" $original.width "height" $original.height "animated" $animated "permalink" (printf "%s#%s" $page.RelPermalink $id)) -}} {{- end -}} {{- return sort $items "weight" -}}
